[NUEVO V5.0] MANDAMIENTO 4: Protocolo Anti-Timeout (La Regla de los 6 Minutos)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Toda función que procese datos DEBE cumplir estas reglas:

4.1 — BATCH OPERATIONS OBLIGATORIAS:
    - PROHIBIDO getValue()/setValue() dentro de bucles.
    - OBLIGATORIO: getValues() UNA vez → procesar en memoria → setValues() UNA vez.
    - Si el rango supera 10,000 celdas, dividir en chunks de 5,000.

    // ✅ CORRECTO
    const data = sheet.getRange(1, 1, lastRow, lastCol).getValues();
    const output = data.map(row => [transform(row[0]), transform(row[1])]);
    sheet.getRange(1, 3, output.length, 2).setValues(output);

    // ❌ PROHIBIDO
    for (let i = 0; i < data.length; i++) {
      sheet.getRange(i+1, 3).setValue(transform(data[i][0]));
    }

4.2 — PAGINACIÓN POR PROGRESO:
    - Si una operación puede exceder 4 minutos (80% del límite), DEBE implementar 
      checkpoint con PropertiesService.
    - Cada lote guarda su progreso: "lastProcessedRow", "lastBatchTimestamp".
    - Si el script se mata, puede reanudar desde el último checkpoint.

    function processLargeDataset() {
      const lock = LockService.getScriptLock();
      lock.waitLock(30000);
      try {
        const props = PropertiesService.getScriptProperties();
        let startRow = parseInt(props.getProperty('lastRow') || '1');
        const BATCH_SIZE = 500;
        const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
        const data = sheet.getRange(startRow, 1, BATCH_SIZE, 5).getValues();
        
        // Procesar lote
        const processed = data.map(row => processRow(row));
        sheet.getRange(startRow, 6, processed.length, 2).setValues(processed);
        
        // Guardar checkpoint
        const nextRow = startRow + data.length;
        if (nextRow <= sheet.getLastRow()) {
          props.setProperty('lastRow', String(nextRow));
          ScriptApp.newTrigger('processLargeDataset').timeBased().after(1000).create();
        } else {
          props.deleteProperty('lastRow'); // Limpieza
        }
      } finally {
        lock.releaseLock();
      }
    }

4.3 — MONITOREO DE TIEMPO INTERNO:
    - Cada función larga DEBE verificar su tiempo de ejecución interno.
    - Si lleva más de 4 minutos, DEBE guardar estado y ceder el control.

    const EXECUTION_START = Date.now();
    const MAX_SAFE_TIME = 240000; // 4 minutos (dejar 2 min de margen)
    
    function isTimeRunningOut() {
      return (Date.now() - EXECUTION_START) > MAX_SAFE_TIME;
    }

Toda operación que MODIFIQUE datos compartidos DEBE usar LockService:

5.1 — NIVELES DE LOCK:
    - getScriptLock(): Para operaciones globales (una sola instancia en todo el script).
    - getDocumentLock(): Para operaciones por documento (permite concurrencia entre docs).

5.2 — PATRÓN OBLIGATORIO:
    function safeUpdate(operation) {
      const lock = LockService.getScriptLock();
      try {
        const acquired = lock.waitLock(30000); // 30 seg timeout
        if (!acquired) throw new Error('LOCK_TIMEOUT: No se pudo adquirir el lock');
        return operation();
      } finally {
        lock.releaseLock();
      }
    }

5.3 — DOBLE VERIFICACIÓN:
    - Después de adquirir el lock, RE-LEER los datos antes de modificar.
    - Otro proceso pudo haberlos cambiado mientras esperabas el lock.

      const today = new Date().toISOString().split('T')[0];
      const counterKey = `emails_sent_${today}`;

7.1 — CACHE OBLIGATORIO PARA DATOS ESTÁTICOS:
    - Todo dato que no cambie cada 5 minutos DEBE cachearse.
    - CacheService tiene TTL máximo de 6 horas (21,600 segundos).

    function getCachedData(key, fetchFunction, ttlSeconds = 3600) {
      const cache = CacheService.getScriptCache();
      let data = cache.get(key);
      
      if (data) {
        return JSON.parse(data);
      }
      
      data = fetchFunction();
      cache.put(key, JSON.stringify(data), ttlSeconds);
      return data;
    }

7.2 — INVALIDACIÓN DE CACHE:

7.3 — CACHE POR USUARIO:

8.1 — URLFETCHAPP CON MANEJO TOTAL DE ERRORES:
    - SIEMPRE usar { muteHttpExceptions: true }.
    - SIEMPRE verificar getResponseCode().
    - SIEMPRE envolver en try/catch.

    function fetchWithRetry(url, options = {}, maxRetries = 3) {
      const defaultOptions = {
        muteHttpExceptions: true,
        followRedirects: true,
        ...options
      };
      
      for (let attempt = 1; attempt <= maxRetries; attempt++) {
        try {
          const response = UrlFetchApp.fetch(url, defaultOptions);
          const code = response.getResponseCode();
          
          if (code >= 200 && code < 300) {
            return { success: true, data: JSON.parse(response.getContentText()) };
          }
          
          if (code === 429) { // Rate limited
            Utilities.sleep(Math.pow(2, attempt) * 1000); // Exponential backoff
            continue;
          }
          
          if (code >= 500) { // Server error, retry
            Utilities.sleep(attempt * 2000);
            continue;
          }
          
          // Client error (4xx), no retry
          return { success: false, error: `HTTP_${code}`, code };
          
        } catch (e) {
          if (attempt === maxRetries) {
            return { success: false, error: e.message };
          }
          Utilities.sleep(attempt * 1000);
        }
      }
      return { success: false, error: 'MAX_RETRIES_EXCEEDED' };
    }

8.2 — BATCH FETCH PARA MÚLTIPLES URLs:
    - Usar UrlFetchApp.fetchAll() cuando haya más de 3 requests al mismo dominio.
    - Reduce tiempo de 3x-10x.

    function fetchMultiple(urls, options = {}) {
      const requests = urls.map(url => ({
        url,
        ...options,
        muteHttpExceptions: true
      }));
      
      const responses = UrlFetchApp.fetchAll(requests);
      return responses.map((resp, i) => ({
        url: urls[i],
        code: resp.getResponseCode(),
        data: resp.getResponseCode() === 200 ? JSON.parse(resp.getContentText()) : null
      }));
    }

    function createTriggerIfNotExists(functionName, type, params) {
      const triggers = ScriptApp.getProjectTriggers();
      const exists = triggers.some(t => t.getHandlerFunction() === functionName);
      
      if (exists) return { created: false, reason: 'ALREADY_EXISTS' };

        case 'time':
          trigger = ScriptApp.newTrigger(functionName).timeBased()

      }
      return { created: true, triggerId: trigger.getUniqueId() };
    }

10.1 — LOGGER ESTRUCTURADO:

    const Logger = {
      _log(level, func, message, data = null) {
        const entry = `[${level}] [${func}] [${new Date().toISOString()}] ${message}`;
        if (data) console.log(entry, JSON.stringify(data));
        else console.log(entry);
        
        // Guardar en Properties para auditoría
        if (level === 'ERROR') {
          const props = PropertiesService.getScriptProperties();
          const errors = JSON.parse(props.getProperty('recent_errors') || '[]');
          errors.push({ level, func, message, time: Date.now() });
          // Mantener solo los últimos 50 errores
          props.setProperty('recent_errors', JSON.stringify(errors.slice(-50)));
        }
      },
      info: (func, msg, data) => Logger._log('INFO', func, msg, data),
      warn: (func, msg, data) => Logger._log('WARN', func, msg, data),
      error: (func, msg, data) => Logger._log('ERROR', func, msg, data),
    };
